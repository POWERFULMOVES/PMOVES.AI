# arm64 (aarch64) hard pins for the media-audio lock compile.
# Rationale: the plain requirements.txt path floats torch on arm64 (observed
# torch 2.13.0 from PyPI against torchaudio 2.11.0+cu128 -> torch._dynamo dies
# at import). These pins encode the SPARK-proven CUDA stack:
#   hi-rag-gateway-v2-gpu on PMOVES-SPARK (GB10) runs torch 2.9.0+cu128 +
#   transformers 4.57.6 healthy for days; media-video's Dockerfile already
#   ARG-pins TORCH_VERSION=2.9.0 for the same reason.
# Wheels verified on download.pytorch.org/whl/cu128 (cp311, manylinux_2_28_aarch64):
#   torch-2.9.0+cu128, torchaudio-2.9.0 (untagged on the cu128 index).
# Install-time wheel SOURCE is made deterministic by the Dockerfile preinstall
# step (--index-url the cu128 index) before the lock install; these versions then
# register as already-satisfied.
torch==2.9.0+cu128
torchaudio==2.9.0
